"""Rank-protected upload page for HMI and oven-temperature images."""

from datetime import date, datetime

import pandas as pd
import streamlit as st

from auth import require_role
from db import FACTORY_TIMEZONE, get_hourly_oven_readings, get_image_uploads, init_db, process_uploaded_oven_image
from nav import render_footer, render_nav

st.set_page_config(page_title="Secretary Uploads | Biashara Masters Sawmill", page_icon="📷", layout="wide")
user = require_role({"Admin", "Secretary"})
render_nav()
init_db()

st.title("Secretary image uploads")
st.caption(f"Signed in as {user['username']} · uploads are shared in the global image container.")
st.info(
    "Upload each oven-temperature image. The app attempts OCR and stores one hourly oven reading per image. "
    "Today's readings are kept and shown below without being tied to a paper roll."
)

with st.form("secretary_image_upload_form", clear_on_submit=True):
    category_label = st.selectbox("Image type", ["Oven temperatures", "HMI"])
    uploads = st.file_uploader(
        "Choose image(s) · JPG, PNG, or WebP · 10 MB maximum each",
        type=["jpg", "jpeg", "png", "webp"],
        accept_multiple_files=True,
    )
    submitted = st.form_submit_button("Upload images", type="primary", use_container_width=True)

if submitted:
    if not uploads:
        st.warning("Select at least one image to upload.")
    else:
        saved = 0
        extracted_count = 0
        for uploaded in uploads:
            try:
                result = process_uploaded_oven_image(
                    uploaded.name,
                    uploaded.type or "",
                    uploaded.getvalue(),
                    user["username"],
                )
                saved += 1
                if result.get("oven_values"):
                    extracted_count += 1
                    st.success(
                        f"Saved {uploaded.name} as an hourly oven reading: "
                        f"{result['oven_values']} for {result['hour_label']}"
                    )
                else:
                    st.warning(
                        f"Image {uploaded.name} was saved, but OCR could not detect six oven values. "
                        f"Review the raw text or upload a clearer scan."
                    )
            except ValueError as error:
                st.error(f"Could not save {uploaded.name}: {error}")
        if saved:
            st.caption(f"Stored {saved} image(s); recognized {extracted_count} hourly oven data capture(s).")
            st.rerun()

st.divider()
st.header("Today's collected oven data")
reading_date = datetime.now(FACTORY_TIMEZONE).date().isoformat()
records = get_hourly_oven_readings(reading_date, limit=24)
if not records:
    st.info("No oven readings have been captured today yet.")
else:
    table = pd.DataFrame(records)
    table = table[[
        "reading_date",
        "hour_label",
        "oven_1",
        "oven_2",
        "oven_3",
        "oven_4",
        "oven_5",
        "oven_6",
        "captured_at",
        "uploaded_by",
    ]]
    st.dataframe(table, use_container_width=True, hide_index=True)

st.divider()
st.header("Global oven-temperature image container")
st.caption("Shared image archive · newest uploads first · retained for today's collected data and review.")
oven_images = get_image_uploads("oven_temperature")
if not oven_images:
    st.info("No oven-temperature images have been uploaded yet.")
else:
    for image in oven_images:
        with st.expander(f"{image['filename']} · uploaded by {image['uploaded_by']} · {image['uploaded_at']} EAT"):
            st.image(image["image_data"], caption=image["filename"], use_container_width=True)
            st.caption(f"Image #{image['id']} · {image['mime_type']} · uploaded by {image['uploaded_by']}")
            if image.get("extracted_numbers"):
                st.code(image["extracted_numbers"], language="json")

st.divider()
st.header("Global HMI image archive")
hmi_images = get_image_uploads("hmi")
if not hmi_images:
    st.info("No HMI images have been uploaded yet.")
else:
    for image in hmi_images:
        with st.expander(f"{image['filename']} · uploaded by {image['uploaded_by']} · {image['uploaded_at']} EAT"):
            st.image(image["image_data"], caption=image["filename"], use_container_width=True)
            st.caption(f"Image #{image['id']} · {image['mime_type']} · uploaded by {image['uploaded_by']}")

render_footer()
